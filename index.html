window.descargarPDF = () => { if (!window.recetaActual) return alert("Selecciona una receta."); const elemento = document.getElementById('zona-impresion'); // TRUCO PARA MÓVILES: Forzar ancho de escritorio temporalmente para evitar 2 páginas const anchoOriginal = elemento.style.width; elemento.style.width = '800px'; const filename = `${window.recetaActual.nombre.replace(/\s+/g, '_')}_${document.getElementById('input-pax').value}${window.recetaActual.unidad_base}.pdf`; const opt = { margin: 0.4, filename, image: { type: 'jpeg', quality: 0.98 }, html2canvas: { scale: 2, useCORS: true, windowWidth: 800 }, jsPDF: { unit: 'in', format: 'letter', orientation: 'portrait' } }; // Generar PDF y luego restaurar el tamaño normal del celular html2pdf().set(opt).from(elemento).save().then(() => { elemento.style.width = anchoOriginal; }); };
